Showcase: How to use Responsive Toolkit

This page showcases the custom breakpoint behavior: include assets, call viewport.use('MyBreakpoints', myBreakpoints) with your custom breakpoints before the document ready event, then query breakpoint aliases.

1) Include JavaScript

<script src="https://code.jquery.com/jquery-3.7.1.min.js"></script>
<script src="responsive-toolkit.min.js"></script>

2) Custom Breakpoints CSS

/*!
 * Custom Breakpoints Demo:
 *
 * custom-1 = mobile (default, < 768px)
 * custom-2 = tablet (768px - 1199px)
 * custom-3 = desktop (>= 1200px)
 */

@layer base, responsive;

@layer base {
    /**
     * Mobile first: default to custom-1 on small screens.
     *
     * Show custom-1 and hide custom-2 and custom-3 when the viewport width is less than 768px.
     */
    .visible-custom-1 {
        display: block;
    }

    .visible-custom-2 {
        display: none;
    }

    .visible-custom-3 {
        display: none;
    }
}

@layer responsive {
    /**
     * Tablet and up: switch visibility to custom-2.
     *
     * Hide custom-1 and show custom-2 when the viewport width is 768px or greater.
     */
    @media (width >= 768px) {
        .visible-custom-1 {
            display: none;
        }

        .visible-custom-2 {
            display: block;
        }
    }

    /**
     * Desktop and up: switch visibility to custom-3.
     *
     * Hide custom-2 and show custom-3 when the viewport width is 1200px or greater.
     */
    @media (width >= 1200px) {
        .visible-custom-2 {
            display: none;
        }

        .visible-custom-3 {
            display: block;
        }
    }
}

3) Do some fun stuff with breakpoints

/* global ResponsiveToolkit */

// Wrap IIFE around your code
(($, viewport) => {
    'use strict';

    // Define custom breakpoints
    const myBreakpoints = {
        'alias-1': $('<div class="device-alias-1 visible-custom-1"></div>'),
        'alias-2': $('<div class="device-alias-2 visible-custom-2"></div>'),
        'alias-3': $('<div class="device-alias-3 visible-custom-3"></div>')
    };

    // Set viewports to custom breakpoints (This needs to be done before the document ready event)
    viewport.use('MyBreakpoints', myBreakpoints);

    $(document).ready(() => {
        // Executes only in alias-1 breakpoint
        if (viewport.is('alias-1')) {
            // …
        }

        // Executes in alias-2 and alias-3 breakpoints
        if (viewport.is('>=alias-2')) {
            // …
        }

        // Executes in alias-1 breakpoint
        if (viewport.is('<alias-2')) {
            // …
        }

        // Execute code each time window size changes
        $(window).resize(
            viewport.changed(() => {
                if (viewport.is('alias-1')) {
                    // …
                }
            })
        );
    });
})(jQuery, ResponsiveToolkit);
