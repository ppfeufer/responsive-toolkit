Showcase: How to use Responsive Toolkit

This page showcases the default toolkit behavior: include assets, do not call viewport.use(...), then query breakpoint aliases.

1) Include CSS and JS

<link rel="stylesheet" href="responsive-toolkit.css">
<script src="https://code.jquery.com/jquery-3.7.1.min.js"></script>
<script src="responsive-toolkit.min.js"></script>

2) Do some fun stuff with breakpoints

/* global ResponsiveToolkit */

// Wrap IIFE around your code
(($, viewport) => {
    'use strict';

    $(document).ready(() => {
        // Executes only in XS breakpoint
        if (viewport.is('xs')) {
            // …
        }

        // Executes in SM, MD and LG breakpoints
        if (viewport.is('>=sm')) {
            // …
        }

        // Executes in XS and SM breakpoints
        if (viewport.is('<md')) {
            // …
        }

        // Execute code each time window size changes
        $(window).resize(
            viewport.changed(() => {
                if (viewport.is('xs')) {
                    // …
                }
            })
        );
    });
})(jQuery, ResponsiveToolkit);

Open the browser console to see breakpoint changes logged on resize.
